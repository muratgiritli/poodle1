// Daily calorie formula for puppies: weight(kg) × 90 kcal (mock simplified)
export function calculateDailyFood(weightKg: number, kcalPer100g: number) {
  const dailyKcal = weightKg * 90;
  const gramsPerDay = Math.round((dailyKcal / kcalPer100g) * 100);
  const mealsPerDay = 3;
  const gramsPerMeal = Math.round(gramsPerDay / mealsPerDay);
  return { dailyKcal, gramsPerDay, mealsPerDay, gramsPerMeal };
}
