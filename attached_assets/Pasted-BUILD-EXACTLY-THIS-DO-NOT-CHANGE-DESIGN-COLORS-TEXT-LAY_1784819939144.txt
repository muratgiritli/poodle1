  selectedDeliveryId: string        // default "standard"
  selectedAddress: string | null
  couponCode: string
  appliedCoupon: CouponCode | null
  showSuccessBanner: boolean        // default true on first load after add

Computed:
  itemCount → sum of quantities
  subtotal → sum(item.salePrice * item.quantity) using ORIGINAL for "Ürünler" line
  productDiscount → sum((original - sale) * quantity)
  shippingCost → IF selectedDelivery express → 79, ELSE IF subtotal >= 500 → 0, ELSE calculate
    Mock: subtotal 1349 >= 500 → shipping FREE (matches mockup "Ücretsiz kargo kazandınız!")
  couponDiscount → appliedCoupon?.discount ?? 0
  total → subtotal + shippingCost - couponDiscount
  totalFormatted → Turkish locale
  installmentAmount → total / 3 formatted "449,67 TL"
  freeShippingProgress → min(subtotal / 500, 1) * 100

Functions:
  updateQuantity(itemId, delta) — min 1, max 10
  removeItem(itemId)
  clearCart()

  addRecommendation(rec) — add to cart + toast
  proceedToPayment() — navigate /odeme or alert("Ödeme sayfası yakında!")
  dismissSuccessBanner()
